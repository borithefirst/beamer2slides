\begin{frame}{A paragraph with a formula}
  Recall that for independent samples the variance of the mean is
  $\operatorname{Var}(\bar X) = \frac{\sigma^2}{n}$, which shrinks as the number of
  samples grows. This is why averaging helps.
\end{frame}

\begin{frame}{Harder cases}
  \begin{theorem}[Cauchy--Schwarz]
    For all vectors $u, v$ we have $|\langle u, v\rangle| \le \sqrt{\langle u,u\rangle}\sqrt{\langle v,v\rangle}$.
  \end{theorem}
  The estimate improves when the sample is large, as shown by
  $\sum_{k=1}^{n} \frac{1}{k^2} < 2$ for all n.
  \[
