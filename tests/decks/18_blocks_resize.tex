% stacked tightly, untitled, with math, lists and tables inside, rounded with shadows and square.

\usetheme{Madrid}
\usepackage{amsmath,booktabs}
\setbeamertemplate{navigation symbols}{}
\title{Blocks under resizing}
\author{Beamer Tester}

\begin{document}

\begin{frame}{Blocks side by side}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{Left}
      A narrow block whose text wraps over a few lines in its column.
    \end{block}
    \column{0.48\textwidth}
    \begin{alertblock}{Right}
      \begin{itemize}
        \item First point
        \item Second point
      \end{itemize}
    \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{Math and tables in blocks}
  \begin{block}{Theorem}

  \end{block}
  \begin{exampleblock}{Results}
    \centering
    \begin{tabular}{lrr}
      \toprule
      Method & Precision & Recall \\
      \midrule
      Ours & 0.89 & 0.83 \\
      \bottomrule
    \end{tabular}
  \end{exampleblock}
\end{frame}
