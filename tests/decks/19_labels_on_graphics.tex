\begin{frame}{Theorems}
  \begin{theorem}[Pythagoras]
    For a right triangle the squares of the legs add up to the square of the hypotenuse.
  \end{theorem}
  \begin{proof}
    Rearrange four copies of the triangle inside a square.
  \end{proof}
  \begin{definition}
    A triangle is right when one angle is ninety degrees.
  \end{definition}
\end{frame}
