\begin{frame}{Inline math inside bullets}
  \begin{itemize}
    \item Let $x \in \mathbb{R}$ and $f(x) = x^2 + 1$.
    \item Pure text bullet with no math at all.
    \item Greek letters $\alpha, \beta, \gamma$ and a fraction $\frac{a}{b}$.
    \item Subscripts: $a_1, a_2, \ldots, a_n$.
  \end{itemize}
\end{frame}

\begin{frame}{Display math}
  The Gaussian integral:
  \[
    \int_{-\infty}^{\infty} e^{-x^2}\,dx = \sqrt{\pi}
  \]
  and an aligned system:
  \begin{align}
    a &= b + c \\
    d &= \sum_{i=1}^{n} i^2
  \end{align}
\end{frame}
