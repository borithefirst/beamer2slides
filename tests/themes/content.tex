\section{Introduction}

\begin{frame}{Motivation}
  \begin{itemize}
    \item Slides written in \LaTeX{} are hard to edit collaboratively
    \item Google Slides is \alert{easy to share}, but has no math typesetting
    \item We want the best of both\footnote{Within reason.}
      \begin{itemize}
        \item Editable text where possible
        \item Pictures where necessary
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Definitions}
  \begin{definition}[Native element]
    An element the user can edit directly in the target application.
  \end{definition}
  \begin{theorem}
    For every deck $D$ there is a conversion $c(D)$ with $|c(D)| = |D|$.
  \end{theorem}
  \begin{proof}
    Obvious.
  \end{proof}
\end{frame}

\section{Method}

\begin{frame}{Pipeline}
  \framesubtitle{From PDF to slides}
  \begin{enumerate}
    \item Extract text spans, images and drawings
    \item Classify: text, pictures, shapes, background
    \item Render the background without converted content
    \item Emit a Google Slides deck
  \end{enumerate}
  \begin{block}{Key idea}
    Measure fidelity on Google's own renderer.
  \end{block}
\end{frame}

\begin{frame}{Math in context}
  The loss is
  \begin{equation}
    \mathcal{L}(\theta) = \sum_{i=1}^{n} \bigl(y_i - f_\theta(x_i)\bigr)^2
  \end{equation}
  where $\theta \in \mathbb{R}^d$ and $n$ is the number of samples.
  \begin{description}
    \item[Inputs] slides as PDF
    \item[Outputs] editable decks
