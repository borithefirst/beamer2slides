\begin{frame}{Why move to the device?}
  \begin{itemize}
    \item Latency: no round trip, answers start in under 100 ms
    \item Privacy: messages never leave the phone
    \item Cost: inference is free once the model is downloaded
    \item Offline: works on the train, in the mountains, anywhere
  \end{itemize}
\end{frame}

\section{The model}

\begin{frame}{Three constraints shaped the model}
  \begin{columns}[T]
    \begin{column}{192\gpt}
      \subhead{Size}
      The model ships inside the app bundle, so it has to stay under 300 MB after
      quantisation.
    \end{column}
    \begin{column}{192\gpt}
      \subhead{Memory}
      Peak RAM must leave room for the rest of the app on a four-year-old mid-range phone.
    \end{column}
    \begin{column}{192\gpt}
      \subhead{Quality}
      Summaries are rated side by side with the cloud model by human raters.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[split]{Quantisation}
  Four-bit weights cut the download by three quarters.

  Quality dropped less than we feared: raters preferred the cloud summary in only a few
  more cases than before.
  \statcard[gblue-pastel]{4×}{smaller download with 4-bit weights}
\end{frame}
